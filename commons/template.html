3. 구현 예시 (Python Code)

가독성이 최적화된 17px Fira Code 환경에서 코드를 분석한다.

# 머신러닝 선형 회귀 샘플 코드
import numpy as np
from sklearn.linear_model import LinearRegression

def train_and_predict():
    # 1. 학습 데이터 생성
    X = np.array([[1], [2], [3], [4]])
    y = np.array([2, 4, 6, 8])

    # 2. 모델 인스턴스화 및 학습
    model = LinearRegression()
    model.fit(X, y)

    # 3. 결과 출력 (Highlight 확인)
    prediction = model.predict([[5]])
    print(f"예측 결과: {prediction}")
    return prediction

if __name__ == "__main__":
    train_and_predict()
